How it works

Generate

Your prompt goes to the model you pick, and AIVAX returns up to four images per request. Each model has its own price per image, and failed images aren't charged. Image generation

Guide

Models that accept references follow up to four pictures: a link, an upload, a pasted screenshot or an image you already made. Styles add a short description to the end of your prompt.

Keep

Collections live in your browser's IndexedDB: only the image links, prompts and costs. The images themselves are hosted by AIVAX at those links.

Build it yourself

One API call: POST /api/v1/generations/images. The models and their prices come from the public GET /api/v1/information/image-models.json. See the API reference or the Worker behind this page. 

# 1. Models, flags (AllowReferenceImage, Deprecated) and list prices. Public, no key needed.
curl https://inference.aivax.net/api/v1/information/image-models.json

# 2. Generate. `count` is 1 to 4 images per prompt. `referenceImages` takes up to four
#    http(s) links or data:image/...;base64 URIs, on models flagged AllowReferenceImage.
curl https://direct.inference.aivax.net/api/v1/generations/images \
  -H "Authorization: Bearer $AIVAX_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "input": "A lighthouse during a winter storm\n\nStyle: anime illustration, cel shading",
    "model": "@black-forest-labs/flux-3-image-klein",
    "count": 2,
    "referenceImages": ["https://example.com/lighthouse.jpg"]
  }'

# → { "data": [ {
#       "input": "A lighthouse during a winter storm...",
#       "images": ["https://.../image-1.webp"],
#       "outputs": [
#         { "url": "https://.../image-1.webp", "cost": 0.019, "error": null },
#         { "url": null, "cost": 0, "error": "The image could not be generated." }   ← not charged

// Works in Node 18+, Bun, Deno and browsers. Keep the API key server-side.
const AIVAX = "https://direct.inference.aivax.net/api/v1";

// Skip deprecated models, and note which ones accept reference images.
async function listModels() {
  const response = await fetch("https://inference.aivax.net/api/v1/information/image-models.json");
  const { data } = await response.json();

  return data
    .filter((model) => !model.flags.includes("Deprecated"))
    .map((model) => ({
      name: model.name,
      label: model.displayName,
      price: model.pricingDescription,
      acceptsReferences: model.flags.includes("AllowReferenceImage"),
    }));
}

async function generateImages({ prompt, model, count = 1, references = [] }) {
  const response = await fetch(`${AIVAX}/generations/images`, {
    method: "POST",
    headers: {
      authorization: `Bearer ${process.env.AIVAX_API_KEY}`,
      "content-type": "application/json",
    },
    body: JSON.stringify({
      input: prompt,
      model,
      count,
      ...(references.length && { referenceImages: references }),
    }),
  });

  const payload = await response.json();
  if (!response.ok) throw new Error(payload.error);

  const [group] = payload.data;       // one group per prompt in `input`
  return { outputs: group.outputs, cost: group.cost };
}

const { outputs, cost } = await generateImages({
  prompt: "A cozy reading nook with a sleeping cat",
  model: "@black-forest-labs/flux-3-image-klein",
  count: 2,
});

for (const output of outputs) console.log(output.url ?? `failed: ${output.error}`);
console.log(`Charged $${cost}`);
// Trimmed from src/worker.js in the repository. The real file adds the AIVAX OAuth
// sign-in that stores each visitor's application key in an HttpOnly cookie.
const AIVAX_API = "https://direct.inference.aivax.net/api/v1";

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    if (url.pathname === "/api/images/generate" && request.method === "POST") {
      const { prompt, model, count, references } = await request.json();

      const response = await fetch(`${AIVAX_API}/generations/images`, {
        method: "POST",
        headers: {
          authorization: `Bearer ${env.AIVAX_API_KEY}`,
          "content-type": "application/json",
        },
        body: JSON.stringify({ input: prompt, model, count, referenceImages: references }),
      });

      const payload = await response.json();
      if (!response.ok) return Response.json({ error: payload.error }, { status: response.status });

      const [group] = payload.data;
      return Response.json({ outputs: group.outputs, cost: group.cost });
    }

    return env.ASSETS.fetch(request);
  },
};
